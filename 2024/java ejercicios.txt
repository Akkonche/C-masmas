package ejercicio.pkg1.guia;

import java.text.DecimalFormat;
import java.util.Scanner;

public class Ejercicio1GUIA 
{
    public static void main(String[] args) {
         DecimalFormat ndec = new DecimalFormat("#0.00");
        Scanner scan = new Scanner(System.in);
        int numeroPersonas;
        double estatura;
        String nombre;
        int edad;
        int adultosMayores = 0;
        int personasBajas = 0;
        char resp;

        do {
            System.out.println("Ingrese el número de personas:");
            numeroPersonas = scan.nextInt();
            scan.nextLine();

            int i = 0;
            do {
                System.out.println("Ingrese el nombre de la persona:");
                nombre = scan.nextLine();
                System.out.println("Ingrese la estatura de la persona (en metros):");
                estatura = scan.nextDouble();
                System.out.println("Ingrese la edad de la persona:");
                edad = scan.nextInt();

                String mensajeEstatura = estatura >= 1.70 ? "Alto" : "Bajo";
                if (mensajeEstatura.equals("Bajo")) {
                    personasBajas++;
                }

                String mensajeEdad;
                if (edad < 13) {
                    mensajeEdad = "Infante";
                } else if (edad >= 13 && edad <= 19) {
                    mensajeEdad = "Adolescente";
                } else if (edad >= 20 && edad <= 59) {
                    mensajeEdad = "Adulto";
                } else {
                    mensajeEdad = "Adulto mayor";
                    adultosMayores++;
                }

                System.out.println("Nombre: " + nombre);
                System.out.println("Estatura: " + ndec.format(estatura) + " metros (" + mensajeEstatura + ")");
                System.out.println("Edad: " + edad + " (" + mensajeEdad + ")");

                i++;
            } while (i < numeroPersonas);

            double porcentajePersonasBajas = ((double) personasBajas / numeroPersonas) * 100;
            System.out.println("Cantidad de adultos mayores: " + adultosMayores);
            System.out.println("Porcentaje de personas bajas: " + ndec.format(porcentajePersonasBajas) + "%");

            System.out.println("Desea continuar? [S/N]");
            resp = scan.next().charAt(0);
            resp = Character.toUpperCase(resp);
            scan.nextLine();
        } while (resp == 'S');
    }
}

            {

                    System.out.println("Error... Ingrese un numero entre 10-50");
            }
            while(vect2[i]<10 ||vect2[i]>50);

            if(vect3[i]>max)
                max=vect3[i];
            
        }
        System.out.println("Datos aleatorios\n");
        for(i=0;i<tam;i++)
            System.out.println(""+ndec.format(vect1[i]));
        
        System.out.println("Datos del valor ingresados\n");
        
        for(i=0;i<tam;i++)
            System.out.println(""+ndec.format(vect2[i]));
        
        System.out.println("Datos Sumados");
        
        for(i=0;i<tam;i++)
            System.out.println(""+ndec.format(vect3[i]));

package ejercicio5;

import java.text.DecimalFormat;
import java.util.Scanner;


public class Ejercicio5 
{
    public static double validar(double valor){
        Scanner scan =new Scanner(System.in);
        while(valor<=0){
            System.out.println("Error... Ingrese un valor positivo");
            valor=scan.nextDouble();
        }
        return valor;
    }
    

    public static void main(String[] args) 
    {
        
       Scanner scan =new Scanner(System.in);
       DecimalFormat ndec=new DecimalFormat("#0.00");
        
       String nombre,obs;
       double peso,estatura,imc;
       int contD=0,contBP=0,contSP=0,cont=0,i;
       
       do{
        System.out.println("Ingrese el nombre: ");
        nombre=scan.nextLine();
        
        System.out.println("Ingrese peso: ");

package matriz;

import java.text.DecimalFormat;
import java.util.Scanner;

public class Matriz {

    public static void main(String[] args) 
    {
       
       DecimalFormat ndec=new DecimalFormat("#.00");
       Scanner scan=new Scanner(System.in);
       
       final int fil=4;
       final int col=3;
       int i,j;
       double peso[][]=new double [fil][col];
       double prom,sum=0,valMin=100;
       
        System.out.println("Ingresando datos a la matriz");
       
        for(i=0;i<fil;i++)
        {
            for(j=0;j<col;j++)
            {
                do
                {
                    System.out.println("Ingrese un peso en la posicion ["+i+"]-["+j+"]");
                    peso[i][j]=scan.nextDouble();
                    
                    if(peso[i][j]<10 || peso[i][j]>99)
                        System.out.println("Error ingrese un peso entre 10 a 99");
                }
                while(peso[i][j]<10||peso[i][j]>99);
                
                sum=sum+=peso[i][j];
                
                if(peso[i][j]<valMin)
                    valMin=peso[i][j];
            }
        }
        
        System.out.println("Datos de la matriz\n");
        for(i=0;i<fil;i++)
        {
            for(j=0;j<col;j++)
            {
                System.out.println(" "+ndec.format(peso[i][j]));
            }
                System.out.println("\n");
        }
        prom=sum/(fil*col);
        System.out.println("El promedio de la Matriz es: "+prom);
        System.out.println("La suma del arreglo es: "+sum);
        System.out.println("Valor minimo de la matriz: "+valMin);
                 
    }
    
    
}
