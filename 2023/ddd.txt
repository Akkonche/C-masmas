/*Escriba un programa en lenguaje C que solicite el nombre de un empleado, sus horas trabajadas completas,su pago por hora,calcular el sueldo bruto.
Al empleado se le deduce el 1.5% de RAP y IHSS asi : Si sueldo bruto es menor que 8000 paga el 5% del sueldo bruto de ihss sino paga L.400.00 Calcular el sueldo neto. 
Limpie pantalla y muestre:Nombre,sueldobruto,ihss,rap,total deducciones y sueldo neto*/
#include<windows.h>
#include<stdio.h>
#include<locale.h>
main()
	{
		system("title Sueldo neto");
		setlocale(LC_ALL,"");
		
		const double CRAP=0.015;
		char nombre[30];
		int horasT;
		double pagoH,sueldoB,rap,ihss,totalD,sueldoN;
		printf("\n\t Calculo de sueldo Neto");
		printf("\n\t ******* ** ****** ****");
		printf("\n Ingrese el nombre del empleado: ");
		scanf ("%s",&nombre);
		printf("\n Ingrese las horas trabajadas: ");
		scanf("%d",&horasT);
		printf("\n Ingrese el pago por hora: ");
		scanf("%lf",pagoH);
		
		sueldoB=horasT*pagoH;
		rap=sueldoB*CRAP;
		/*if (sueldoB<8000)
			ihss=sueldoB*0.05;
		else
			ihss=400;*/
			ihss=(sueldoB<8000)?(400):(sueldoB*0.05);
			
		totalD=rap+ihss;
		sueldoN=sueldoB-totalD;
		system("cls");
		printf("\n\t Calculo de sueldo Neto");
		printf("\n\t ******* ** ****** ****");
		printf("\n\t Empleado: %s",nombre);
		printf("\n\t Sueldo bruto: %.2lf", sueldoB);
		printf("\n\t Deduccion ihss: %.2lf",ihss);
		printf("\n\t Deduccion RAP: %.2lf",rap);
		printf("\n\t Total Deducciones: %.2lf",totalD);
		printf("\n\t Sueldo Neto: %.2lf",sueldoN);
		return 0;
	}

					printf("\n Error, Ingrese un numero entre 10 y 50");
			}

		}
